name: Build Android APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download 3D model packs
        run: |
          set -e
          mkdir -p assets/models/zombies assets/models/player assets/models/city /tmp/packs
          cd /tmp/packs
          for f in zombie-walker zombie-runner zombie-brute zombie-crawler; do
            wget -q "https://vinish-models.pages.dev/zombies/$f.glb" -O "$GITHUB_WORKSPACE/assets/models/zombies/$f.glb"
          done
          wget -q https://vinish-models.pages.dev/people/soldier.glb -O "$GITHUB_WORKSPACE/assets/models/player/soldier.glb"
          wget -q https://github.com/KayKit-Game-Assets/KayKit-City-Builder-Bits-1.0/archive/refs/heads/main.zip -O city.zip
          unzip -q city.zip "KayKit-City-Builder-Bits-1.0-main/addons/kaykit_city_builder_bits/Assets/gltf/*"
          cp KayKit-City-Builder-Bits-1.0-main/addons/kaykit_city_builder_bits/Assets/gltf/* "$GITHUB_WORKSPACE/assets/models/city/"
          find "$GITHUB_WORKSPACE/assets/models" -type f | sort
          du -sh "$GITHUB_WORKSPACE/assets/models"

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - name: Download Godot 4.3 and export templates
        run: |
          wget -q https://github.com/godotengine/godot/releases/download/4.3-stable/Godot_v4.3-stable_linux.x86_64.zip
          unzip -q Godot_v4.3-stable_linux.x86_64.zip
          wget -q https://github.com/godotengine/godot/releases/download/4.3-stable/Godot_v4.3-stable_export_templates.tpz
          mkdir -p ~/.local/share/godot/export_templates/4.3.stable
          unzip -q Godot_v4.3-stable_export_templates.tpz -d /tmp/tpz
          cp -r /tmp/tpz/templates/* ~/.local/share/godot/export_templates/4.3.stable/

      - name: Point Godot at the Android SDK
        run: |
          mkdir -p ~/.config/godot
          printf '[gd_resource type="EditorSettings" format=3]\n\n[resource]\nexport/android/android_sdk_path = "%s"\nexport/android/java_sdk_path = "%s"\n' "$ANDROID_SDK_ROOT" "$JAVA_HOME" > ~/.config/godot/editor_settings-4.tres

      - name: Create debug keystore
        run: |
          mkdir -p ~/.android
          keytool -genkeypair -v -keystore ~/.android/debug.keystore -alias androiddebugkey -storepass android -keypass android -keyalg RSA -keysize 2048 -validity 10000 -dname "CN=Android Debug,O=Android,C=US"

      - name: Import project resources
        run: ./Godot_v4.3-stable_linux.x86_64 --headless --editor --quit || true

      - name: Install Android build template into project
        run: |
          mkdir -p android/build
          unzip -q -o ~/.local/share/godot/export_templates/4.3.stable/android_source.zip -d android/build
          echo "4.3.stable" > android/.build_version
          touch android/build/.gdignore
          ls -la android android/build

      - name: Export Android APK
        env:
          GODOT_ANDROID_KEYSTORE_DEBUG_PATH: /home/runner/.android/debug.keystore
          GODOT_ANDROID_KEYSTORE_DEBUG_USER: androiddebugkey
          GODOT_ANDROID_KEYSTORE_DEBUG_PASSWORD: android
        run: |
          mkdir -p build
          ./Godot_v4.3-stable_linux.x86_64 --headless --export-debug "Android" build/zombie-survival.apk

      - name: Align and re-sign APK
        run: |
          BT=$(ls -d "${ANDROID_HOME:-$ANDROID_SDK_ROOT}"/build-tools/* | sort -V | tail -1)
          "$BT/zipalign" -f -P 16 4 build/zombie-survival.apk build/aligned.apk || "$BT/zipalign" -f -p 4 build/zombie-survival.apk build/aligned.apk
          "$BT/apksigner" sign --ks ~/.android/debug.keystore --ks-key-alias androiddebugkey --ks-pass pass:android --key-pass pass:android --v1-signing-enabled true --v2-signing-enabled true --v3-signing-enabled true --out build/zombie-survival.apk build/aligned.apk
          "$BT/apksigner" verify build/zombie-survival.apk

      - name: Publish APK release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: test-apk-${{ github.run_number }}
          name: Test APK ${{ github.run_number }}
          files: build/zombie-survival.apk
